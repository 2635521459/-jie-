/* 本地复现：目录与用户片段拼接后，规范化可离开 sandbox。
 * 只计算路径字符串，不 fopen，不访问设备。
 */
#include <stdio.h>
#include <string.h>

static int normalize(const char *in, char *out, size_t cap) {
    const char *part_ptr[48];
    size_t part_len[48];
    int count = 0;
    const char *p = in;
    int absolute = (*in == '/');
    size_t used = 0;
    int i;

    while (*p != '\0') {
        const char *start;
        size_t n;
        while (*p == '/' || *p == '\\') {
            p++;
        }
        if (*p == '\0') {
            break;
        }
        start = p;
        while (*p != '\0' && *p != '/' && *p != '\\') {
            p++;
        }
        n = (size_t)(p - start);
        if (n == 1 && start[0] == '.') {
            continue;
        }
        if (n == 2 && start[0] == '.' && start[1] == '.') {
            if (count > 0) {
                count--;
            }
            continue;
        }
        if (count >= 48) {
            return -1;
        }
        part_ptr[count] = start;
        part_len[count] = n;
        count++;
    }

    if (cap == 0) {
        return -1;
    }
    out[0] = '\0';
    if (absolute) {
        out[0] = '/';
        out[1] = '\0';
        used = 1;
    }
    for (i = 0; i < count; i++) {
        if (used > 1 || (used == 1 && out[0] != '/')) {
            if (used + 1 >= cap) {
                return -1;
            }
            out[used++] = '/';
            out[used] = '\0';
        }
        if (used + part_len[i] >= cap) {
            return -1;
        }
        memcpy(out + used, part_ptr[i], part_len[i]);
        used += part_len[i];
        out[used] = '\0';
    }
    if (used == 0) {
        out[0] = '.';
        out[1] = '\0';
    }
    return 0;
}

int main(void) {
    const char *base = <<BASE_LIT>>;
    const char *user = "../other_app/secret.txt";
    char joined[512];
    char norm[512];
    char sandbox[512];
    size_t slen;
    int inside;
    int escaped;

    snprintf(joined, sizeof(joined), "%s/%s", base, user);
    if (normalize(joined, norm, sizeof(norm)) != 0) {
        return 1;
    }
    if (normalize(base, sandbox, sizeof(sandbox)) != 0) {
        return 1;
    }
    slen = strlen(sandbox);
    inside = strcmp(norm, sandbox) == 0 ||
             (strncmp(norm, sandbox, slen) == 0 && norm[slen] == '/');
    escaped = !inside;
    printf("joined=%s\n", joined);
    printf("normalized=%s\n", norm);
    printf("sandbox=%s\n", sandbox);
    printf("escaped=%s\n", escaped ? "yes" : "no");
    if (!escaped) {
        printf("cwe22 concat not triggered\n");
        return 1;
    }
    printf("cwe22 concat triggered\n");
    return 0;
}
